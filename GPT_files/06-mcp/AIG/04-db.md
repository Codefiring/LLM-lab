 Database Design — mcp_scan.db (SQLite)

     Schema

     PRAGMA journal_mode = WAL;    -- better read/write concurrency
     PRAGMA foreign_keys = ON;

     -- LLM service profiles (replaces llm_profiles.json)
     CREATE TABLE IF NOT EXISTS llm_profiles (
         id          TEXT PRIMARY KEY,
         name        TEXT NOT NULL UNIQUE,
         base_url    TEXT NOT NULL,
         api_key     TEXT NOT NULL,      -- stored in plaintext (local tool)
         model       TEXT NOT NULL,
         is_default  INTEGER NOT NULL DEFAULT 0,   -- 0/1 boolean
         created_at  TEXT NOT NULL                 -- ISO 8601
     );

     -- Top-level scan task (one per "开始扫描" click)
     CREATE TABLE IF NOT EXISTS tasks (
         id                  TEXT PRIMARY KEY,
         status              TEXT NOT NULL DEFAULT 'pending',  -- pending|running|completed|failed
         created_at          TEXT NOT NULL,
         completed_at        TEXT,
         llm_profile_id      TEXT,           -- SET NULL if profile later deleted
         llm_profile_name    TEXT NOT NULL,  -- snapshot at task creation
         llm_model           TEXT NOT NULL,  -- snapshot at task creation
         prompt              TEXT NOT NULL DEFAULT '',
         language            TEXT NOT NULL DEFAULT 'zh',
         has_oauth           INTEGER NOT NULL DEFAULT 0,
         current_target_idx  INTEGER NOT NULL DEFAULT 0,
         log_file            TEXT,           -- absolute path, set after subprocess starts
         error               TEXT,
         FOREIGN KEY (llm_profile_id) REFERENCES llm_profiles(id) ON DELETE SET NULL
     );

     -- Per-target within a task (one row per URL)
     CREATE TABLE IF NOT EXISTS task_targets (
         id          TEXT PRIMARY KEY,
         task_id     TEXT NOT NULL,
         position    INTEGER NOT NULL,       -- 0-based insertion order
         name        TEXT,                   -- optional display name (from config)
         url         TEXT NOT NULL,
         stage_ids   TEXT,                   -- JSON array e.g. "[2,5,11]"; NULL = all 15
         status      TEXT NOT NULL DEFAULT 'pending',   -- pending|running|completed|failed
         score       INTEGER,                -- 0-100; NULL until completed
         readme      TEXT,                   -- Markdown info-collection report
         started_at  TEXT,
         ended_at    TEXT,
         error       TEXT,
         FOREIGN KEY (task_id) REFERENCES tasks(id) ON DELETE CASCADE
     );

     -- Per-stage progress for each target scan
     CREATE TABLE IF NOT EXISTS task_stages (
         id              TEXT PRIMARY KEY,
         task_target_id  TEXT NOT NULL,
         stage_id        INTEGER NOT NULL,   -- 1–27
         name            TEXT NOT NULL,
         status          TEXT NOT NULL DEFAULT 'pending',  -- pending|running|completed|error
         output          TEXT NOT NULL DEFAULT '',         -- Markdown from Final Output log lines
         updated_at      TEXT,
         FOREIGN KEY (task_target_id) REFERENCES task_targets(id) ON DELETE CASCADE
     );

     -- Individual vulnerability findings per target
     CREATE TABLE IF NOT EXISTS vulnerabilities (
         id              TEXT PRIMARY KEY,
         task_target_id  TEXT NOT NULL,
         title           TEXT NOT NULL,
         description     TEXT,              -- Markdown
         risk_type       TEXT,              -- e.g. "Command Injection", "CWE-78", "MCP01"
         level           TEXT,              -- "High"/"高危"/"Medium"/"中危" etc.
         suggestion      TEXT,              -- Markdown fix recommendations
         FOREIGN KEY (task_target_id) REFERENCES task_targets(id) ON DELETE CASCADE
     );

     Indexes

     CREATE INDEX IF NOT EXISTS idx_tasks_created    ON tasks(created_at DESC);
     CREATE INDEX IF NOT EXISTS idx_tasks_status     ON tasks(status);
     CREATE INDEX IF NOT EXISTS idx_targets_task     ON task_targets(task_id, position);
     CREATE INDEX IF NOT EXISTS idx_stages_target    ON task_stages(task_target_id, stage_id);
     CREATE INDEX IF NOT EXISTS idx_vulns_target     ON vulnerabilities(task_target_id);

     DB Helper

     from contextlib import contextmanager

     @contextmanager
     def get_db():
         conn = sqlite3.connect(DB_FILE)
         conn.row_factory = sqlite3.Row
         conn.execute("PRAGMA journal_mode=WAL")
         conn.execute("PRAGMA foreign_keys=ON")
         try:
             yield conn
             conn.commit()
         except Exception:
             conn.rollback()
             raise
         finally:
             conn.close()

     All DB writes use parameterized queries (? placeholders). No string interpolation.

     Startup / Recovery

     def init_db():
         with get_db() as conn:
             conn.executescript(SCHEMA_SQL)   # creates tables + indexes if not exist
             # Mark any tasks that were 'running' at last shutdown as 'failed'
             conn.execute(
                 "UPDATE tasks SET status='failed', error='Server restarted' "
                 "WHERE status IN ('pending', 'running')"
             )

     ├────────────────────┼──────────────────────────────────────────────────────────────────────────────────────────────────┤
     │ List tasks         │ SELECT t.*, COUNT(tt.id) as target_count FROM tasks t LEFT JOIN task_targets tt ON               │

     ├────────────────────┼──────────────────────────────────────────────────────────────────────────────────────────────────┤

     │ status             │                                                                                                  │

     monitor_task → DB Write Flow

     The monitor_task background thread:
     1. Runs LogParser on each new log line
     2. On stage start event → UPDATE task_stages SET status='running'
     3. On stage complete event → UPDATE task_stages SET status='completed', output=?
     4. On target complete (Stage 27 done):
       - INSERT INTO vulnerabilities (...) for each result
       - UPDATE task_targets SET status='completed', score=?, readme=?
       - UPDATE tasks SET current_target_idx=?
     5. On process exit → UPDATE tasks SET status=?, completed_at=?
     6. Each write also puts an SSE event into sse_queues[task_id]

     SSE Stream — Live Event Queue

     # SSE generator reads from asyncio.Queue populated by monitor thread
     async def task_stream(task_id: str):
         queue = asyncio.Queue()
         sse_queues[task_id] = queue
         try:
             # Send full current state first (catch-up for reconnects)
             task = db_get_task_full(task_id)
             yield f"event: init\ndata: {json.dumps(task)}\n\n"
             while True:
                 try:
                     event = await asyncio.wait_for(queue.get(), timeout=15.0)
                     if event is None:  # sentinel: scan done
                         break
                     yield f"event: {event['type']}\ndata: {json.dumps(event['data'])}\n\n"
                 except asyncio.TimeoutError:
                     yield ": keep-alive\n\n"
         finally:
             sse_queues.pop(task_id, None)

     monitor_task writes to the queue via: asyncio.run_coroutine_threadsafe(queue.put(event), event_loop).

     The init event on reconnect lets the frontend restore the full task state from DB, so no events are lost.
