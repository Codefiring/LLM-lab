digraph KVM_IOCTL_STATE_MACHINE {
    rankdir=LR;
    node [shape=ellipse];

    ### -------------------------
    ### /dev/kvm 设备层状态
    ### -------------------------
    subgraph cluster_kvmdev {
        label = "/dev/kvm 层";
        style = "rounded,dashed";

        KDEV_CLOSED   [label="KDEV_CLOSED\n(未 open)"];
        KDEV_OPEN     [label="KDEV_OPEN\n(open 成功)"];
        KDEV_READY    [label="KDEV_READY\n(API/扩展检查完成)"];

        KDEV_CLOSED -> KDEV_OPEN [label="open(\"/dev/kvm\")"];
        KDEV_OPEN -> KDEV_READY  [label="KVM_GET_API_VERSION\nKVM_CHECK_EXTENSION*"];

        // 从 /dev/kvm 创建 VM
        KDEV_READY -> VM_NEW [label="KVM_CREATE_VM", color="blue"];
    }

    ### -------------------------
    ### VM fd 层状态
    ### -------------------------
    subgraph cluster_vm {
        label = "VM fd 层";
        style = "rounded,dashed";

        VM_NEW         [label="VM_NEW\n(刚创建的 VM)"];
        VM_MEM_CFG     [label="VM_MEM_CFG\n(内存已配置)"];
        VM_IRQCHIP_CFG [label="VM_IRQCHIP_CFG\n(中断/PIT/TSS 等配置)"];
        VM_READY_VCPU  [label="VM_READY_VCPU\n(可创建 vCPU)"];
        VM_RUNNING     [label="VM_RUNNING\n(存在 vCPU\nenabled)"];
        VM_DESTROYED   [label="VM_DESTROYED\n(关闭 VM fd)"];

        // 内存相关
        VM_NEW     -> VM_MEM_CFG [label="KVM_SET_USER_MEMORY_REGION*"];

        // 中断、时钟等可选配置
        VM_MEM_CFG -> VM_IRQCHIP_CFG [label="KVM_CREATE_IRQCHIP\nKVM_CREATE_PIT2\nKVM_SET_TSS_ADDR\nKVM_SET_CLOCK*"];

        // 实际上只要内存配置正确就可以创建 vCPU
        VM_MEM_CFG     -> VM_READY_VCPU  [label="(内存 OK)"];
        VM_IRQCHIP_CFG -> VM_READY_VCPU  [label="(内存 + IRQ/PIT/TSS OK)"];

        // VM fd 被关闭
        VM_READY_VCPU -> VM_DESTROYED [label="close(vm_fd)"];
        VM_RUNNING    -> VM_DESTROYED [label="close(vm_fd)"];
    }

    ### -------------------------
    ### vCPU fd 层状态
    ### -------------------------
    subgraph cluster_vcpu {
        label = "vCPU fd 层";
        style = "rounded,dashed";

        VCPU_NONE       [label="VCPU_NONE\n(尚未创建 vCPU)"];
        VCPU_CREATED    [label="VCPU_CREATED\n(创建 vCPU)\n(未 mmap)"];
        VCPU_MMAP_RUN   [label="VCPU_MMAP_RUN\n(kvm_run 已 mmap)"];
        VCPU_CONFIGURED [label="VCPU_CONFIGURED\n(CPUID/MSR/寄存器配置)"];
        VCPU_RUNNING    [label="VCPU_RUNNING\n(KVM_RUN 循环中)"];
        VCPU_STOPPED    [label="VCPU_STOPPED\n(退出 KVM_RUN)"];
        VCPU_CLOSED     [label="VCPU_CLOSED\n(关闭 vcpu_fd)"];

        // 从 VM_READY_VCPU 创建 vCPU
        VM_READY_VCPU -> VCPU_NONE [label="", style="dotted"];

        VCPU_NONE    -> VCPU_CREATED  [label="KVM_CREATE_VCPU"];
        VCPU_CREATED -> VCPU_MMAP_RUN [label="KVM_GET_VCPU_MMAP_SIZE\nmmap(vcpu_fd)"];

        // 配置 vCPU 状态（顺序在这里抽象为“都要完成”）
        VCPU_MMAP_RUN -> VCPU_CONFIGURED [label="KVM_SET_CPUID2\nKVM_SET_MSRS*\nKVM_SET_REGS\nKVM_SET_SREGS"];

        // 运行循环
        VCPU_CONFIGURED -> VCPU_RUNNING  [label="KVM_RUN"];
        VCPU_RUNNING    -> VCPU_STOPPED  [label="VMEXIT / signal\n(KVM_RUN 返回)"];
        VCPU_STOPPED    -> VCPU_RUNNING  [label="KVM_RUN (再次进入)", dir="both"];

        // 关闭 vCPU
        VCPU_STOPPED -> VCPU_CLOSED [label="close(vcpu_fd)"];
        VCPU_RUNNING -> VCPU_CLOSED [label="close(vcpu_fd)\n(通常先令其退出)"];
    }

    ### VM 与 vCPU 之间的关系（抽象）
    VM_READY_VCPU -> VCPU_CREATED [label="KVM_CREATE_VCPU\n(可多次，多个 vCPU)", color="blue"];
    VCPU_CREATED  -> VM_RUNNING   [label="存在至少一个 vCPU\n(抽象提升为 VM_RUNNING)", style="dashed"];
    VCPU_CLOSED   -> VM_READY_VCPU [label="最后一个 vCPU 关闭\n回到仅 VM 存在的状态", style="dashed"];
}

digraph loop_state_machine {
    rankdir=LR;
    node [shape=ellipse, fontsize=12];

    // ----- States -----
    UNBOUND               [label="UNBOUND\nlo_state=Lo_unbound\nno backing_file"];
    BOUND_RO              [label="BOUND_RO\nLo_bound + READ_ONLY"];
    BOUND_RW              [label="BOUND_RW\nLo_bound + READ_WRITE"];
    BOUND_RO_AUTOCLEAR    [label="BOUND_RO_AUTOCLEAR\nBOUND_RO + LO_FLAGS_AUTOCLEAR"];
    BOUND_RW_AUTOCLEAR    [label="BOUND_RW_AUTOCLEAR\nBOUND_RW + LO_FLAGS_AUTOCLEAR"];
    DELETING              [label="DELETING\nLo_deleting (optional sink)"];

    // ----- Attach / configure -----
    // losetup / LOOP_SET_FD / LOOP_CONFIGURE 成功绑定
    UNBOUND -> BOUND_RW [
        label="LOOP_SET_FD / LOOP_CONFIGURE\n(backing file + writable mode)\nlo_state: Lo_unbound -> Lo_bound\nlo_flags: maybe clear RO"
    ];

    UNBOUND -> BOUND_RO [
        label="LOOP_SET_FD / LOOP_CONFIGURE\n(backing file read-only)\nlo_state: Lo_unbound -> Lo_bound\nlo_flags |= LO_FLAGS_READ_ONLY"
    ];

    // 配置信息变更（只在已绑定上合法）
    BOUND_RW -> BOUND_RW [
        label="LOOP_SET_STATUS64 / LOOP_SET_CAPACITY /\nLOOP_CHANGE_FD (same mode)\nupdate offset/sizelimit/flags\nbacking_file may change"
    ];
    BOUND_RO -> BOUND_RO [
        label="LOOP_SET_STATUS64 / LOOP_SET_CAPACITY /\nLOOP_CHANGE_FD (same mode)\nupdate offset/sizelimit/flags\nbacking_file may change"
    ];

    // 读写属性切换（通过 LOOP_SET_STATUS64 修改 flags）
    BOUND_RW -> BOUND_RO [
        label="LOOP_SET_STATUS64\nset LO_FLAGS_READ_ONLY"
    ];
    BOUND_RO -> BOUND_RW [
        label="LOOP_SET_STATUS64\nclear LO_FLAGS_READ_ONLY\n(backing file & open mode allow write)"
    ];

    // ----- I/O 操作 -----
    // 设备 I/O 实际上只是自环，不改变状态
    BOUND_RW -> BOUND_RW [
        label="read / write / mmap\n正常读写后端文件"
        style=solid;
    ];

    BOUND_RO -> BOUND_RO [
        label="read / mmap\n(write -> error)"
        style=solid;
    ];

    BOUND_RW_AUTOCLEAR -> BOUND_RW_AUTOCLEAR [
        label="read / write / mmap\n直到最后一个 fd 关闭"
        style=solid;
    ];

    BOUND_RO_AUTOCLEAR -> BOUND_RO_AUTOCLEAR [
        label="read / mmap\n直到最后一个 fd 关闭"
        style=solid;
    ];

    // ----- LOOP_CLR_FD 手动 detach -----
    // 正常 detach：没有其他打开的 fd，直接清除绑定
    BOUND_RW -> UNBOUND [
        label="LOOP_CLR_FD\nif disk_openers == 1\n-> __loop_clr_fd()\nlo_state -> Lo_unbound\nclear backing_file/flags"
    ];

    BOUND_RO -> UNBOUND [
        label="LOOP_CLR_FD\nif disk_openers == 1\n-> __loop_clr_fd()\nlo_state -> Lo_unbound"
    ];

    // 有其他打开者：只设置 AUTOCLEAR，真正清理在最后一次 close
    BOUND_RW -> BOUND_RW_AUTOCLEAR [
        label="LOOP_CLR_FD\nif disk_openers > 1\nlo_flags |= LO_FLAGS_AUTOCLEAR\n(不立刻清理)"
    ];

    BOUND_RO -> BOUND_RO_AUTOCLEAR [
        label="LOOP_CLR_FD\nif disk_openers > 1\nlo_flags |= LO_FLAGS_AUTOCLEAR\n(不立刻清理)"
    ];

    // ----- AUTOCLEAR + 最后一次 close -----
    // 对应 loop_release() 中的逻辑：最后一个打开者关闭时调用 __loop_clr_fd()
    BOUND_RW_AUTOCLEAR -> UNBOUND [
        label="last close() on /dev/loopX\n(lo_flags & AUTOCLEAR)\n-> __loop_clr_fd()\nlo_state -> Lo_unbound"
    ];

    BOUND_RO_AUTOCLEAR -> UNBOUND [
        label="last close() on /dev/loopX\n(lo_flags & AUTOCLEAR)\n-> __loop_clr_fd()\nlo_state -> Lo_unbound"
    ];

    // ----- open/close 不改变配置状态 -----
    // 单纯 open/close 只改变引用计数，对状态机来说是自环
    UNBOUND -> UNBOUND [
        label="open()/close()\n(不能做 I/O / 配置，大多 ioctl 失败)"
        style=dashed;
    ];

    BOUND_RW -> BOUND_RW [
        label="open()/close() (refcnt +/-)\n不影响绑定关系"
        style=dashed;
    ];

    BOUND_RO -> BOUND_RO [
        label="open()/close() (refcnt +/-)\n不影响绑定关系"
        style=dashed;
    ];

    // AUTOCLEAR 状态下的 open/close 同样只影响计数
    BOUND_RW_AUTOCLEAR -> BOUND_RW_AUTOCLEAR [
        label="open()/close() except last\n只更新 refcnt"
        style=dashed;
    ];

    BOUND_RO_AUTOCLEAR -> BOUND_RO_AUTOCLEAR [
        label="open()/close() except last\n只更新 refcnt"
        style=dashed;
    ];

    // ----- 设备删除路径（可选） -----
    // /dev/loop-control + LOOP_CTL_REMOVE 或模块卸载时
    UNBOUND  -> DELETING [label="loop_remove() / module_exit\n(no backing, safe remove)" style=dotted];
    BOUND_RO -> DELETING [label="loop_remove() after clr_fd\n(理论上先转 UNBOUND 再删除)" style=dotted];
    BOUND_RW -> DELETING [label="loop_remove() after clr_fd\n(同上)" style=dotted];

    // 常见图形化布局小美化
    { rank=same; BOUND_RO; BOUND_RW; }
    { rank=same; BOUND_RO_AUTOCLEAR; BOUND_RW_AUTOCLEAR; }
}
